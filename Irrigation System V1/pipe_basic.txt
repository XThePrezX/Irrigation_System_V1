#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>

#define BUFSIZE 256

int main() {
    int fd[2];
    char *messages[] = {
        "Hello from parent",
        "Pipes are unidirectional",
        "Data flows one way",
        "Kernel buffers the bytes",
        "EOF signals end of stream"
    };

    /* TODO 1: Create the pipe */
    /* Hint: pipe(fd) --- fd[0] is read end, fd[1] is write end */
    pipe(fd);

    pid_t pid = fork();

    if (pid < 0) {
        perror("fork"); exit(1);
    }

    if (pid == 0) {
        /* === CHILD: reads from pipe === */

        /* TODO 2: Close the write end (child doesn't write) */
        close(fd[1]);

        char buf[BUFSIZE];
        ssize_t n;
        while ((n = read(fd[0], buf, BUFSIZE - 1)) > 0) {
            buf[n] = '\0';
            printf("[Child] Received: %s", buf);
        }

        /* TODO 3: Close the read end when done */
        close(fd[0]);

        exit(0);

    } else {
        /* === PARENT: writes to pipe === */

        /* TODO 4: Close the read end (parent doesn't read) */
        close(fd[0]);

        for (int i = 0; i < 5; i++) {
            char line[BUFSIZE];
            snprintf(line, BUFSIZE, "Message %d: %s\n", i + 1, messages[i]);
            /* TODO 5: Write line to the pipe */
            write(fd[1], line, strlen(line));

        }

        /* TODO 6: Close the write end (signals EOF to child) */
        close(fd[1]);

        wait(NULL);
        printf("[Parent] Child finished.\n");
    }

    return 0;
}
